You need your own firebase-config.js file to import firebase config.

EXAMPLE: "firebase-config.js"
const firebaseConfig = {
  apiKey: ...,
  authDomain: ...,
  databaseURL: ...,
  projectId: ...,
  storageBucket: ...,
  messagingSenderId: ...,
  appId: ...,
  measurementId: ...
};

export default firebaseConfig;
